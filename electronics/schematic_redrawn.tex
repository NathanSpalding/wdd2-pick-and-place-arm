\newcommand{\chip}[9]{%
  \node[chip=#4cm, minimum height={(#7+1)*\pp cm}] (#1) at (#2,#3) {};
  \node[ref, above=1pt of #1.north, align=center] {#8\ \normalfont\scriptsize #9};
  \foreach \lab [count=\i] in {#5} {
    \draw (#1.north west) ++(0,-\pp*\i) coordinate (#1-L\i) -- ++(-0.35,0) coordinate (#1-l\i);
    \node[pinl] at (#1-L\i) {\lab};}
  \foreach \lab [count=\i] in {#6} {
    \draw (#1.north east) ++(0,-\pp*\i) coordinate (#1-R\i) -- ++(0.35,0) coordinate (#1-r\i);
    \node[pinr] at (#1-R\i) {\lab};}
}
